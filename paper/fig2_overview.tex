\node[anchor=north west, font=\scriptsize\bfseries, text=blue!60!black] at (5.75,6.0)
  {MCP gateway with STPG (inserted on the write / read / sink paths)};

\node[module, anchor=south west] (PPT) at (0.3,2.15) {%
  \textcolor{blue!60!black}{\textbf{PPT} (on write)}\\
  \textcolor{blue!60!black}{Persistent Provenance Tagging}\\[3pt]
  $\bullet$ sidecar $\{$source id, writer, $\tau$, $\pi\}$\\
  $\bullet$ lineage: \texttt{src:}$\langle$connector$\rangle$ $|$ \texttt{write:}id\\
  $\bullet$ persists across sessions and stores\\
  $\bullet$ never reads ground-truth labels};

\node[module, anchor=south west] (TSA) at (5.75,2.15) {%
  \textcolor{blue!60!black}{\textbf{TSA} (on read)}\\
  \textcolor{blue!60!black}{Temporal Shard Assembly}\\[3pt]
  $\bullet$ group by lineage, not by window\\
  $\bullet$ reassemble $s_1\cdots s_k$ across sessions\\
  $\bullet$ scorer: embedding, then judge\\
  $\bullet$ $E(G)$ per lineage; $\mathcal{G}$, $F$ to the gate};

\node[module, anchor=south west] (PDG) at (11.25,2.15) {%
  \textcolor{blue!60!black}{\textbf{PDG} (on sink)}\\
  \textcolor{blue!60!black}{Privilege-Delta Gate}\\[3pt]
  $\bullet$ bind call destination $\delta$ to lineages\\
  $\bullet$ witness: does the lineage target $\delta$?\\
  $\bullet$ trusted: pass; untrusted: block\\
  $\bullet$ no evidence $\to$ block; $\Delta\pi$ logged};
